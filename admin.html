async function loadAlerts() { const RENDER_URL = "https://disaster-api-af2w.onrender.com/api/all-alerts"; const response = await fetch(RENDER_URL); const alerts = await response.json(); let tableBody = document.getElementById("alertTable"); tableBody.innerHTML = ""; alerts.forEach(alert => { tableBody.innerHTML += ` ${alert.name} ${alert.location} ${new Date(alert.timestamp).toLocaleString()} `; }); }
